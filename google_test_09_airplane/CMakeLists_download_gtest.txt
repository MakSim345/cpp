cmake_minimum_required(VERSION 3.14) # check the version of CMake

# Give the name to executable:
# NOTE: ".exe" will be added in win32 automatically.
set(PROJECT_NAME runTests09)
project(${PROJECT_NAME})

# GoogleTest requires at least C++14
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Ensure MSVC-specific flags are set
if (MSVC)
    add_compile_options(/W4 /permissive-) # Enable warnings and strict standard conformance
endif()

#==============================================================================
# Disable SSL verification for git (temporary fix)
set(GIT_SSL_NO_VERIFY ON CACHE BOOL "Disable SSL verification for git" FORCE)

include(FetchContent)
FetchContent_Declare(
  googletest
  GIT_REPOSITORY https://github.com/google/googletest.git
  GIT_TAG        release-1.10.0
)
# For Windows: Prevent overriding the parent project compiler/linker settings
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)

#==============================================================================

enable_testing()

# Add source files
set(sources
        main_plane_test.cpp
        airplane.cpp
        error_message.cpp
        plane.cpp
        utils.cpp
    )

# Add the executable
add_executable(${PROJECT_NAME} ${sources})

# Link GoogleTest libraries
target_link_libraries(
    ${PROJECT_NAME}
    gtest_main
    gtest
)

# Ensure proper runtime library linkage on MSVC
if (MSVC)
    target_compile_options(${PROJECT_NAME} PRIVATE /MTd) # Use static runtime for debug builds
endif()

# Include GoogleTest for test discovery
include(GoogleTest)
gtest_discover_tests(${PROJECT_NAME})
